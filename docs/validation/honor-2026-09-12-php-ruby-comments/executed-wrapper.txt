]

results = []

def run(label, args, cwd):
    record = {
        "label": label,
        "cwd": str(cwd),
        "command": [str(x) for x in args],
        "selectedEnv": {key: env.get(key) for key in selected_keys},
    }
    (ROOT / f"{label}-pre-env.json").write_text(
        json.dumps(record["selectedEnv"], indent=2), encoding="utf-8"
    )
    (ROOT / f"{label}-command.json").write_text(
        json.dumps({"cwd": record["cwd"], "command": record["command"]}, indent=2),
        encoding="utf-8",
    )
    proc = subprocess.run(
        record["command"], cwd=str(cwd), env=env,
        stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
        text=True, encoding="utf-8", errors="replace",
    )
    (ROOT / f"{label}.log").write_text(proc.stdout, encoding="utf-8")
    record["exitCode"] = proc.returncode
    record["log"] = f"{label}.log"
    (ROOT / f"{label}-result.json").write_text(
        json.dumps(record, indent=2), encoding="utf-8"
    )
    results.append(record)
    print(f"{label}: {proc.returncode}", flush=True)
    return proc.returncode

def stop_if_failed(label, code):
    if code != 0:
        (ROOT / "summary.json").write_text(json.dumps(results, indent=2), encoding="utf-8")
        raise SystemExit(f"{label} failed with exit code {code}")
