$Utf8NoBom = [Text.UTF8Encoding]::new($false)

function Quote-Display([string]$Value) {
  if ($Value -match '[\s\"'']') {
    return '"' + $Value.Replace('"', '\"') + '"'
  }
  return $Value
}

function Invoke-Captured {
  param(
    [Parameter(Mandatory)][string]$Name,
    [Parameter(Mandatory)][string]$WorkingDirectory,
    [Parameter(Mandatory)][string]$Executable,
    [Parameter(Mandatory)][string[]]$Arguments
  )

  $script:CommandIndex++
  $Prefix = '{0:D2}-{1}' -f $script:CommandIndex, $Name
  $StdoutPath = Join-Path $LogRoot ($Prefix + '.stdout.log')
  $StderrPath = Join-Path $LogRoot ($Prefix + '.stderr.log')
  $SnapshotPath = Join-Path $LogRoot ($Prefix + '.env.json')
  $ExitPath = Join-Path $LogRoot ($Prefix + '.exitcode.txt')
  $CommandPath = Join-Path $LogRoot ($Prefix + '.command.txt')
  $DisplayCommand = (($Executable) + ' ' + (($Arguments | ForEach-Object { Quote-Display $_ }) -join ' ')).Trim()

  $Snapshot = [ordered]@{
    timestamp = [DateTimeOffset]::Now.ToString('o')
    command = $DisplayCommand
    workingDirectory = $WorkingDirectory
    USERPROFILE = $env:USERPROFILE
    HOME = $env:HOME
    APPDATA = $env:APPDATA
    LOCALAPPDATA = $env:LOCALAPPDATA
    CODEX_HOME = $env:CODEX_HOME
    XDG_CONFIG_HOME = $env:XDG_CONFIG_HOME
    TMP = $env:TMP
    TEMP = $env:TEMP
    NPM_CONFIG_CACHE = $env:NPM_CONFIG_CACHE
    NPM_CONFIG_USERCONFIG = $env:NPM_CONFIG_USERCONFIG
    NPM_CONFIG_GLOBALCONFIG = $env:NPM_CONFIG_GLOBALCONFIG
    UAGENT_SYNC_WORKSPACE_ROOT = [Environment]::GetEnvironmentVariable('UAGENT_SYNC_WORKSPACE_ROOT', 'Process')
    OPENCODE_SYNC_WORKSPACE_ROOT = [Environment]::GetEnvironmentVariable('OPENCODE_SYNC_WORKSPACE_ROOT', 'Process')
  }
  [IO.File]::WriteAllText($SnapshotPath, ($Snapshot | ConvertTo-Json -Depth 4), $Utf8NoBom)
  [IO.File]::WriteAllText($CommandPath, $DisplayCommand, $Utf8NoBom)

  Push-Location $WorkingDirectory
  try {
    & $Executable @Arguments 1> $StdoutPath 2> $StderrPath
    $ExitCode = $LASTEXITCODE
  }
  catch {
    [IO.File]::AppendAllText($StderrPath, ($_ | Out-String), $Utf8NoBom)
    $ExitCode = if ($LASTEXITCODE) { $LASTEXITCODE } else { 9001 }
  }
  finally {
    Pop-Location
  }

  [IO.File]::WriteAllText($ExitPath, [string]$ExitCode, $Utf8NoBom)
  $script:CommandRecords += [ordered]@{
    index = $script:CommandIndex
    name = $Name
    command = $DisplayCommand
    workingDirectory = $WorkingDirectory
    exitCode = $ExitCode
    stdout = [IO.Path]::GetFileName($StdoutPath)
    stderr = [IO.Path]::GetFileName($StderrPath)
    environment = [IO.Path]::GetFileName($SnapshotPath)
  }
  Write-Host ("[{0}] exit={1} {2}" -f $Prefix, $ExitCode, $DisplayCommand)
  return $ExitCode
}
