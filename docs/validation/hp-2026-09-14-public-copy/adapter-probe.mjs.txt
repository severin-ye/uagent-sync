import fs from 'node:fs';import assert from 'node:assert/strict';import path from 'node:path';import cp from 'node:child_process';import {syncBuiltinESMExports} from 'node:module';import {createHash} from 'node:crypto';
import {loadVerifiedPublicSkills} from './src/scripts/verified-public-profile.mjs';
const base='<CONTROLLED_ROOT>',root='<USER_HOME>/hpr/src',real=cp.spawnSync;
const manifest=JSON.parse(fs.readFileSync(root+'/data/reviewed-public-skills.json'));const rows=[];
try{let calls=[];cp.spawnSync=(command,args)=>{calls.push({command,operation:args.includes('cat-file')?'cat-file':args[0]});return{status:1,stdout:Buffer.alloc(0)};};syncBuiltinESMExports();assert.throws(()=>loadVerifiedPublicSkills(base+'/probe-fetch-failed'),/Git operation failed/);assert.equal(calls.length,1);rows.push({case:'download failure throws before provider exists',pass:true,calls:calls.length});
 calls=[];cp.spawnSync=(command,args)=>{calls.push(args);return{status:0,stdout:args.includes('cat-file')?Buffer.from('ARTIFICIAL_WRONG_BYTES'):Buffer.alloc(0)};};syncBuiltinESMExports();assert.throws(()=>loadVerifiedPublicSkills(base+'/probe-hash-failed'),/checksum mismatch/);assert.equal(calls.length,3);assert.ok(calls[1].includes('fetch.negotiationAlgorithm=noop'));assert.ok(calls[1].includes('--filter=blob:none'));rows.push({case:'fixed digest mismatch throws; promisor semantics retained',pass:true,calls:calls.length});
 assert.throws(()=>loadVerifiedPublicSkills(base+'/public-cache'),/new absolute cache/);rows.push({case:'existing cache rejected',pass:true});
}finally{cp.spawnSync=real;syncBuiltinESMExports();}
// Independently re-read only the fetched Git blobs, never a local Skill copy.
const repos=[...new Set(manifest.files.map(f=>f.repository))],verified=[];let previous=0;
for(const repository of repos){const entries=manifest.files.filter(f=>f.repository===repository),repo=base+'/public-cache/repo-'+previous+'.git';for(const f of entries){let b=cp.execFileSync('git',['-C',repo,'cat-file','blob',f.blob],{maxBuffer:40e6});if(f.lineEndings==='crlf')b=Buffer.from(new TextDecoder('utf-8',{fatal:true,ignoreBOM:true}).decode(b).replaceAll('\n','\r\n'));assert.equal(createHash('sha256').update(b).digest('hex'),f.sha256);verified.push({source:f.source,verified:true,lineEndings:f.lineEndings});}previous+=entries.length;}
const result={artificial:rows,verifiedPublicBlobs:verified.length,files:verified,realNpmInvoked:false,noPersonalSourceForProvider:true};fs.writeFileSync('<USER_HOME>/hpr/adapter-review.json',JSON.stringify(result,null,2));console.log(JSON.stringify({artificial:rows,verifiedPublicBlobs:verified.length}));
