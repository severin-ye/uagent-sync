    'USERPROFILE', 'HOME', 'APPDATA', 'LOCALAPPDATA', 'CODEX_HOME',
    'XDG_CONFIG_HOME', 'TMP', 'TEMP', 'NPM_CONFIG_CACHE',
    'NPM_CONFIG_USERCONFIG', 'NPM_CONFIG_GLOBALCONFIG',
    'UAGENT_SYNC_WORKSPACE_ROOT', 'OPENCODE_SYNC_WORKSPACE_ROOT',
    'OPENCODE_CONFIG_DIR', 'OPENCODE_HOME'
)
$script:Sequence = 0
$script:Results = @()

function Save-EnvironmentSnapshot([string]$Prefix) {
    $snapshot = [ordered]@{}
    foreach ($name in $TrackedEnv) {
        $item = Get-Item -LiteralPath ("Env:" + $name) -ErrorAction SilentlyContinue
        $snapshot[$name] = if ($null -eq $item) { $null } else { $item.Value }
    }
    $snapshot | ConvertTo-Json -Depth 3 | Out-File -LiteralPath ($Prefix + '.env.json') -Encoding utf8
}

function Invoke-Logged([string]$Label, [string]$Display, [string]$Exe, [string[]]$Arguments, [string]$WorkingDirectory) {
    $script:Sequence++
    $safeLabel = $Label -replace '[^A-Za-z0-9._-]', '_'
    $prefix = Join-Path $Logs ('{0:D2}-{1}' -f $script:Sequence, $safeLabel)
    Save-EnvironmentSnapshot $prefix
    $Display | Out-File -LiteralPath ($prefix + '.command.txt') -Encoding utf8
    $WorkingDirectory | Out-File -LiteralPath ($prefix + '.cwd.txt') -Encoding utf8
    Push-Location -LiteralPath $WorkingDirectory
    try {
        & $Exe @Arguments 1> ($prefix + '.stdout.log') 2> ($prefix + '.stderr.log')
        $exitCode = $LASTEXITCODE
    }
    finally {
        Pop-Location
    }
    if ($null -eq $exitCode) { $exitCode = 0 }
    [string]$exitCode | Out-File -LiteralPath ($prefix + '.exit.txt') -Encoding ascii
    $script:Results += [pscustomobject]@{
        sequence = $script:Sequence
        label = $Label
        command = $Display
        cwd = $WorkingDirectory
        exitCode = $exitCode
        stdout = ($prefix + '.stdout.log')
        stderr = ($prefix + '.stderr.log')
        environment = ($prefix + '.env.json')
    }
    $script:Results | ConvertTo-Json -Depth 5 | Out-File -LiteralPath (Join-Path $RunRoot 'command-results.json') -Encoding utf8
    if ($exitCode -ne 0) { throw "Command failed ($exitCode): $Display" }
}
